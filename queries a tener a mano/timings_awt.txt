--#tabla transitions (todas)
,transitions as (
  select *
  ,lag(row_state) over (partition by order_code order by state_created_at) previous_row_state  
  from (
  select order_code
  ,timestamp_diff(state_created_at, previous_state_created_at,minute) minutes_diff
  ,* except(order_code) 
  ,lag(previous_state) over(partition by order_code order by previous_state_created_at) before_previous_state
   from (
    select created_date_local
    ,safe_cast(l.platform_order_code as int64) order_code 
    ,l.vendor.vertical_type
   ,round(l.estimated_prep_time/60,0) estimated_preptime
   ,d.rider_near_restaurant_at_local 
   ,d.delivery_status
   ,lag(datetime(t.created_at, "America/Santiago")) over (partition by platform_order_code order by t.created_at) previous_state_created_at
    ,datetime(t.created_at, "America/Santiago") state_created_at
    ,lag(t.state) over (partition by platform_order_code order by t.created_at) previous_state
    ,t.state    
    ,lead(t.state) over (partition by platform_order_code order by t.created_at) next_state
    ,lead(t.actions.issue_type) over (partition by platform_order_code order by t.created_at) next_issue
  ,t.rider_id
  ,t.actions.performed_by 
  ,row_number() over(partition by platform_order_code, t.state order by t.created_at asc) row_state --#nro de vez que sucede el estado
  ,t.undispatch_type 
  ,t.actions.issue_type  
  ,t.actions.reason 
  ,t.estimated_pickup_arrival_at_local estimated_pu_arrival 
  ,t.estimated_dropoff_arrival_at_local estimated_do_arrival
   from `peya-bi-tools-pro.il_logistics.fact_logistic_orders` l
  left join unnest(l.deliveries) as d
  left join unnest(d.transitions) as t
  where l.country_code='cl'
  and l.created_date_local between  dinf  and current_date()-1
  and d.is_primary
  and l.peya_order_id is not null 
  order by order_code, t.created_at asc ) )
 )

--#cantidad de undispatches por orden 
,und as ( 
select order_code
,case when delivery_status = "completed" and couriers_notified<total_manual_undispatch+1 then total_manual_undispatch+1
when delivery_status ="cancelled" and couriers_notified<total_manual_undispatch then total_manual_undispatch 
else couriers_notified end couriers_notified,
total_manual_undispatch,
reason_LatePreparation
 from (
select order_code
,delivery_status
,count(case when state="courier_notified" then rider_id end) couriers_notified
,count(case when undispatch_type in ("temp_offline_undispatch","manual_undispatch") then rider_id end) total_manual_undispatch  
,count(CASE WHEN undispatch_type='manual_undispatch' AND performed_by='issue_service' AND issue_type IN ('Issue::DispatchCourierRedispatchRequest') and reason IN ('Late Order Preparation') THEN rider_id end) as reason_LatePreparation
from transitions 
group by 1,2 
  )
)

            SUM(IF(reject_reason = 'FRAUD_PRANK', FR_OTHER_NON_ORDER_RELATED, 0)) AS FRAUD_PRANK,

            SUM(IF(reject_reason = 'NO_COURIER', FR_OTHER_NON_ORDER_RELATED, 0)) AS NIU_NO_COURIER,

    -- DISTANCES
    SUM(DISTANCES) AS DISTANCES, 
        SUM(TR_HIGH_DISTANCE + TR_HIGH_PREP_TIME_AND_DISTANCE / 2 + TR_HIGH_PREP_TIME_AND_DISTANCE_COMBINATION / 2) AS DRIVING_TIME_IMPACT,
